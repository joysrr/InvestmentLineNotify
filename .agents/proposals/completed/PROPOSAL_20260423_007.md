```javascript
// 純函式 export 供測試
export function parseFrontmatter(content) {
  const match = content.match(/^---\n([\s\S]*?)\n---/);
  if (!match) return null;
  const fm = {};
  for (const line of match[1].split('\n')) {
    const m = line.match(/^([a-z_]+):\s*(.+)$/);
    if (m) fm[m[1]] = m[2].replace(/^["']|["']$/g, '');
  }
  return fm;
}

export function buildVersionTable(filesInfo) {
  const lines = ['| 文件 | 當前版本 | 最後修改日期 | 最近提案 | 審計基準版本 |'];
  lines.push('|------|---------|------------|---------|------------|');
  for (const { path, version, last_modified, last_proposal, audit_baseline } of filesInfo) {
    lines.push(`| \`${path}\` | **${version}** | ${last_modified} | ${last_proposal} | ${audit_baseline || '—'} |`);
  }
  return lines.join('\n');
}

// CLI entry guard
const isMainModule = import.meta.url === `file://${process.argv[1]}`;
if (isMainModule) {
  const filesInfo = TRACKED_FILES.map(rel => {
    const abs = join(AGENTS_ROOT, '..', rel);
    const content = readFileSync(abs, 'utf-8');
    const fm = parseFrontmatter(content);
    return {
      path: rel,
      version: fm?.version || '<unknown>',
      last_modified: fm?.last_modified || '<unknown>',
      last_proposal: fm?.last_proposal || '<unknown>',
      audit_baseline: fm?.audit_baseline,
    };
  });
  const table = buildVersionTable(filesInfo);
  console.log(table);
  // --write 模式：直接更新 RULES_VERSION_INDEX.md 對應區塊
  if (process.argv.includes('--write')) {
    const indexPath = join(AGENTS_ROOT, 'RULES_VERSION_INDEX.md');
    const indexContent = readFileSync(indexPath, 'utf-8');
    // 找 「## 版本總覽」與下一個 `## ` 之間的內容並替換
    const updated = indexContent.replace(
      /(## 版本總覽\n\n)([\s\S]*?)(\n## )/,
      `$1${table}\n$3`
    );
    writeFileSync(indexPath, updated);
    console.log('✅ RULES_VERSION_INDEX.md 已更新');
  }
}
```

```javascript
// scripts/generators/gen-proposal-index.mjs

export function scanProposals(activeDir, completedDir, archivedDir) {
  const active = readdirSync(activeDir).filter(f => f.endsWith('.md') && f !== '.gitkeep');
  const completed = readdirSync(completedDir).filter(f => f.endsWith('.md') && f !== '.gitkeep');
  const archived = readdirSync(archivedDir).filter(f => f.endsWith('.md') && f !== '.gitkeep');

  // 解析 active/ 各檔狀態（OPEN / IN_PROGRESS）
  const openProposals = [];
  const inProgressProposals = [];
  for (const f of active) {
    const content = readFileSync(join(activeDir, f), 'utf-8');
    const status = parseProposalStatus(content);
    const meta = parseProposalMetadata(content);
    if (status === 'OPEN') openProposals.push(meta);
    else if (status === 'IN_PROGRESS') inProgressProposals.push(meta);
  }

  // 已完成 / 取消 / 歸檔 數量
  const cancelled = completed.filter(f => isCancelled(f));  // 解析狀態
  const done = completed.filter(f => !isCancelled(f));

  return {
    counts: {
      OPEN: openProposals.length,
      IN_PROGRESS: inProgressProposals.length,
      DONE: done.length,
      CANCELLED: cancelled.length,
      ARCHIVED: archived.length,
      total: openProposals.length + inProgressProposals.length + completed.length + archived.length,
    },
    openProposals,
    inProgressProposals,
  };
}
```

```javascript
// scripts/generators/gen-framework-version.mjs

export function bumpVersion(current, type = 'minor') {
  // 解析 'v4.5' → { major: 4, minor: 5 }
  const m = current.match(/^v(\d+)\.(\d+)$/);
  if (!m) throw new Error(`Invalid version: ${current}`);
  const major = parseInt(m[1], 10);
  const minor = parseInt(m[2], 10);
  if (type === 'minor') return `v${major}.${minor + 1}`;
  if (type === 'major') return `v${major + 1}.0`;
  throw new Error(`Unknown bump type: ${type}`);
}

baseline_proposal: ${baselineProposal}

proposals_included:
  total: ${proposalsTotal}
  framework: ${proposalsTotal}

last_included_proposal: ${lastProposal}

auditor_baseline:
  last_audit_date: ${releaseDate}
  last_audit_score: ${auditScore}
  drift_score: ${driftScore}

notes: |
${notes.split('\n').map(l => '  ' + l).join('\n')}
`;
}

    if (args[i] === '--bump' && args[i + 1]) bumpType = args[++i];
    else if (args[i] === '--proposal' && args[i + 1]) proposalId = args[++i];
    else if (args[i] === '--write') writeMode = true;
  }
  // 1. 讀當前 FRAMEWORK_VERSION
  // 2. 讀 PROPOSAL_INDEX 統計（由 gen-proposal-index 邏輯）
  // 3. bump version + 更新欄位
  // 4. console.log 預覽 / --write 寫入
}
```
