## 一、問題描述

### 1.1 User 痛點（2026-04-23）

User 真實跑 `init-workspace.mjs` smoke test 發現 **2 個 bug**，耗費多輪除錯才釐清：

1. **Bug 1（邏輯 bug）**：頂層單欄位 block 處理錯誤
   - 例：`FUNCTIONAL_DOMAINS` schema 結構為 `{ type: 'table', required, description }` 直接置於頂層
   - 原 `collectAnswers` 迴圈誤把 `type`/`required`/`description` 當子欄位逐一詢問
   - 若有單元測試測 `parseYaml` + `collectAnswers` 遍歷邏輯，本 bug 應被立即發現

2. **Bug 2（環境 bug）**：`readline/promises.question()` 在 non-TTY 下只消費一行後 silently close
   - 已知 Node.js `readline/promises` 限制：pipe/heredoc 輸入時 question() 第一次後 readline 即 close
   - 症狀：CLI 走到第二個 prompt 後 hang 至 EOF；timeout 後 exit 0（看似正常但未完成任務）
   - 若有 E2E smoke test（piped input）本 bug 應被立即發現

### 1.2 缺口量化

| 層面 | 當前狀態 |
|-----|---------|
| `scripts/framework-diff.mjs`（~200 行）| 僅 self-diff 手動驗證；無函式單元測試 |

| 風險 | 機率 | 後果 |

#### 3.1 目標

#### 3.2 Bash 框架設計

```bash
#!/bin/bash
# smoke.test.sh — scripts/ E2E 煙霧測試
# 依 PROPOSAL_20260423_006 建立

set -u
SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
AGENTS_ROOT="$(cd "$SCRIPT_DIR/../.." && pwd)"
TEMP_DIR=""  # 測試前建立；最後清理
PASS_COUNT=0
FAIL_COUNT=0
declare -a FAILED_TESTS=()

# -------- Helpers --------
setup() {
  TEMP_DIR="$(mktemp -d /tmp/agents-scripts-test.XXXXXX)"
  # 備份 CURRENT_WORKSPACE（若修改到需還原）
  cp "$AGENTS_ROOT/CURRENT_WORKSPACE" "$TEMP_DIR/CURRENT_WORKSPACE.backup" 2>/dev/null || true
}

cleanup() {
  # 還原 CURRENT_WORKSPACE
  if [ -f "$TEMP_DIR/CURRENT_WORKSPACE.backup" ]; then
    cp "$TEMP_DIR/CURRENT_WORKSPACE.backup" "$AGENTS_ROOT/CURRENT_WORKSPACE"
  fi
  # 清除所有測試 workspace（防測試失敗殘留）
  rm -rf "$AGENTS_ROOT/workspaces/smoke_test_"*
  rm -rf "$AGENTS_ROOT/workspaces/smoke_rollback_"*
  rm -f "$AGENTS_ROOT/workspaces/smoke_rollback_"* 2>/dev/null || true
  rm -rf "$TEMP_DIR"
}

trap cleanup EXIT

pass() { echo "✅ PASS: $1"; PASS_COUNT=$((PASS_COUNT+1)); }
fail() { echo "❌ FAIL: $1 — $2"; FAIL_COUNT=$((FAIL_COUNT+1)); FAILED_TESTS+=("$1"); }

# -------- 測試案例 --------

# Test 1: init-workspace --help
test_help() {
  local out
  out=$(node "$AGENTS_ROOT/scripts/init-workspace.mjs" --help 2>&1)
  if echo "$out" | grep -q "Interactive workspace initialization"; then
    pass "init-workspace --help"
  else
    fail "init-workspace --help" "expected '--help' output"
  fi
}

# Test 2: init-workspace --dry-run 不寫檔
test_dry_run_no_write() {
  local before after
  before=$(ls "$AGENTS_ROOT/workspaces/" | sort)
  # 提供最小輸入（觸發 pre-check 確認 → 跳過 schema 問答 via dry-run early exit 不實用）
  # 改用完整輸入但 dry-run
  echo "y" | timeout 10 node "$AGENTS_ROOT/scripts/init-workspace.mjs" --name smoke_test_dry --dry-run --force < /dev/null >/dev/null 2>&1 || true
  after=$(ls "$AGENTS_ROOT/workspaces/" | sort)
  if [ "$before" = "$after" ]; then
    pass "init-workspace --dry-run（不寫檔）"
  else
    fail "init-workspace --dry-run" "檔案系統變更"
  fi
}

# Test 3: init-workspace 完整成功路徑
test_success_path() {
  local name="smoke_test_success"
  local input=$(generate_minimal_input)  # helper function
  echo "$input" | timeout 15 node "$AGENTS_ROOT/scripts/init-workspace.mjs" --name "$name" --force >/dev/null 2>&1
  local exit_code=$?

  if [ $exit_code -ne 0 ]; then
    fail "完整成功路徑" "CLI exit $exit_code"
    return
  fi
  if [ ! -f "$AGENTS_ROOT/workspaces/$name.md" ]; then
    fail "完整成功路徑" "workspaces/$name.md 未建立"; return
  fi
  if [ ! -d "$AGENTS_ROOT/workspaces/$name/proposals/active" ]; then
    fail "完整成功路徑" "proposals/active 目錄未建立"; return
  fi
  if ! grep -q "^$name$" "$AGENTS_ROOT/CURRENT_WORKSPACE"; then
    fail "完整成功路徑" "CURRENT_WORKSPACE 未更新"; return
  fi
  pass "完整成功路徑（$name）"
}

# Test 4: init-workspace rollback 路徑（預建檔案佔位觸發 mkdirSync fail）
test_rollback_path() {
  local name="smoke_rollback_test"
  touch "$AGENTS_ROOT/workspaces/$name"  # 建立同名檔案佔位
  local input=$(generate_minimal_input)
  echo "$input" | timeout 15 node "$AGENTS_ROOT/scripts/init-workspace.mjs" --name "$name" --force >/dev/null 2>&1

  # 驗證 rollback
  if [ -f "$AGENTS_ROOT/workspaces/$name.md" ]; then
    fail "rollback" "workspaces/$name.md 未被 rollback"
  fi
  if [ -d "$AGENTS_ROOT/workspaces/$name/proposals" ]; then
    fail "rollback" "proposals 子樹未清除"
  fi
  rm -f "$AGENTS_ROOT/workspaces/$name"  # 清除佔位檔
  pass "rollback 路徑（mkdirSync fail → 完整回退）"
}

# Test 5: framework-diff self-diff exit 0
test_framework_diff_self() {
  node "$AGENTS_ROOT/scripts/framework-diff.mjs" --target "$AGENTS_ROOT" >/dev/null 2>&1
  if [ $? -eq 0 ]; then
    pass "framework-diff self-diff exit 0"
  else
    fail "framework-diff self-diff" "非零退出碼"
  fi
}

# Test 6: init-workspace CURRENT_WORKSPACE 保護（無 --force 時回 "n" 取消）
test_current_workspace_guard() {
  local before
  before=$(cat "$AGENTS_ROOT/CURRENT_WORKSPACE")
  local out
  out=$(echo "n" | timeout 5 node "$AGENTS_ROOT/scripts/init-workspace.mjs" --name smoke_test_cancel 2>&1)
  local after
  after=$(cat "$AGENTS_ROOT/CURRENT_WORKSPACE")
  if [ "$before" = "$after" ] && echo "$out" | grep -q "使用者取消"; then
    pass "CURRENT_WORKSPACE 保護（無 --force 回 n 取消）"
  else
    fail "CURRENT_WORKSPACE 保護" "取消流程異常"
  fi
}

# -------- Main --------
setup
test_help
test_dry_run_no_write
test_success_path
test_rollback_path
test_framework_diff_self
test_current_workspace_guard

echo ""
echo "========================================"
echo "Smoke Tests: $PASS_COUNT passed / $FAIL_COUNT failed"
if [ $FAIL_COUNT -gt 0 ]; then
  echo "Failed tests:"
  for t in "${FAILED_TESTS[@]}"; do echo "  - $t"; done
  exit 1
fi
echo "========================================"
exit 0
```
