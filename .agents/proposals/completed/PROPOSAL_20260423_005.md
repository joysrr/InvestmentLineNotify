#### 1.4 核心程式架構

```javascript
#!/usr/bin/env node
// init-workspace.mjs
// 依 PROPOSAL_20260423_005 建立；純 Node.js 內建模組
// 對齊 tasks/project-onboarding.md 6 步驟 SOP

import { createInterface } from "node:readline/promises";
import {
  readFileSync,
  existsSync,
  writeFileSync,
  mkdirSync,
  readdirSync,
  rmSync,

import { join, dirname, resolve } from "node:path";
import { fileURLToPath } from "node:url";
import { spawnSync } from "node:child_process";

const __dirname = dirname(fileURLToPath(import.meta.url));
const AGENTS_ROOT = resolve(join(__dirname, ".."));
const SCHEMA_PATH = join(AGENTS_ROOT, "WORKSPACE_SCHEMA.yaml");
const CURRENT_WORKSPACE_PATH = join(AGENTS_ROOT, "CURRENT_WORKSPACE");
const WORKSPACES_DIR = join(AGENTS_ROOT, "workspaces");
const TECH_STACKS_DIR = join(AGENTS_ROOT, "tech-stacks");
const DEFAULT_ENV_PATH = join(WORKSPACES_DIR, "default_env.md");

// -------- CLI args --------
const args = process.argv.slice(2);
let nameArg = null,
  dryRun = false,
  force = false,
  help = false;
for (let i = 0; i < args.length; i++) {
  if (args[i] === "--name" && args[i + 1]) nameArg = args[++i];
  else if (args[i] === "--dry-run") dryRun = true;
  else if (args[i] === "--force") force = true;
  else if (args[i] === "--help" || args[i] === "-h") help = true;
}

if (help) {
  printHelp();
  process.exit(0);
}

const rl = createInterface({ input: process.stdin, output: process.stdout });

// -------- Schema 解析（沿用 workspace-schema-validator 的 parseYaml）--------
function parseYaml(text) {
  /* 同 workspace-schema-validator.mjs */
}

// -------- Step 0: 前置檢查 --------
async function preCheck() {
  if (!existsSync(SCHEMA_PATH)) fatal(`Schema not found: ${SCHEMA_PATH}`);
  if (existsSync(CURRENT_WORKSPACE_PATH) && !force) {
    const current = readFileSync(CURRENT_WORKSPACE_PATH, "utf-8").trim();
    const ans = await rl.question(
      `⚠️  CURRENT_WORKSPACE = "${current}"；確定建立新 workspace 並切換？[y/N] `,
    );
    if (!ans.toLowerCase().startsWith("y")) process.exit(0);
  }
  const schema = parseYaml(readFileSync(SCHEMA_PATH, "utf-8"));
  const availableTechStacks = readdirSync(TECH_STACKS_DIR)
    .filter((f) => f.endsWith(".md"))
    .map((f) => f.replace(".md", ""));
  return { schema, availableTechStacks };
}

// -------- Step 1-5: 逐 schema 區塊問答 --------
async function collectAnswers(schema, availableTechStacks) {
  const answers = {};
  const topBlocks = Object.keys(schema).filter(
    (k) => !["schema_version", "last_modified", "last_proposal"].includes(k),
  );

  for (const block of topBlocks) {
    console.log(`\n=== ${block} ===`);
    const blockFields = schema[block];
    for (const [field, def] of Object.entries(blockFields)) {
      const hint = def.required ? "[required]" : "[optional]";
      const desc = def.description ? `\n  ${def.description}` : "";
      let ans = await rl.question(`${field} ${hint}:${desc}\n> `);

      // 熔斷檢查（TECH_STACK 區塊）
      if (
        block === "TECH_STACK" &&
        ans &&
        !answerMatchesAvailableStacks(ans, availableTechStacks)
      ) {
        const choice = await rl.question(
          `⚠️  "${ans}" 無對應 tech-stacks/*.md。[1] 中止 [2] 跳過該技術棧規範 [3] 修改輸入: `,
        );
        if (choice === "1") fatal("使用者中止 onboarding");
        if (choice === "3") {
          ans = await rl.question(`${field} ${hint}: `);
        }
        // choice === '2' → 繼續（採通用規範）
      }

      // required 阻斷
      if (def.required && !ans.trim()) {
        fatal(`required 欄位 "${block}.${field}" 未填寫；無法繼續`);
      }

      answers[`${block}.${field}`] = ans.trim() || "<not_applicable>";
    }
  }
  return answers;
}

async function previewAndCommit(name, answers) {

  const proposalsDir = join(wsDir, "proposals");
  const content = renderWorkspaceMd(name, answers);

  console.log("\n📋 將建立：");
  console.log(`  - ${wsFile}`);
  console.log(`  - ${wsDir}/proposals/{active,completed,archived}/.gitkeep`);
  console.log(`  - ${wsDir}/proposals/PROPOSAL_INDEX.md（空殼）`);
  console.log(`  - ${CURRENT_WORKSPACE_PATH}（寫入 ${name}）`);

  if (dryRun) {
    console.log("\n[DRY-RUN] 不執行實際寫檔");
    return;
  }

  const ok = await rl.question("\n確認執行？[y/N] ");
  if (!ok.toLowerCase().startsWith("y")) process.exit(0);

  // 寫檔（transaction-like）
  try {
    mkdirSync(join(proposalsDir, "active"), { recursive: true });
    mkdirSync(join(proposalsDir, "completed"), { recursive: true });
    mkdirSync(join(proposalsDir, "archived"), { recursive: true });
    writeFileSync(join(proposalsDir, "active", ".gitkeep"), "");
    writeFileSync(join(proposalsDir, "completed", ".gitkeep"), "");
    writeFileSync(join(proposalsDir, "archived", ".gitkeep"), "");
    writeFileSync(
      join(proposalsDir, "PROPOSAL_INDEX.md"),
      renderProposalIndexShell(name),
    );
    writeFileSync(wsFile, content);
    writeFileSync(CURRENT_WORKSPACE_PATH, name + "\n");
  } catch (e) {
    console.error(`❌ 寫檔失敗：${e.message}`);
    rollback(name);
    process.exit(2);
  }

  // 驗證
  const validateRes = spawnSync(
    "node",
    [
      join(AGENTS_ROOT, "tests", "workspace-schema-validator.mjs"),
      "--workspace",
      name,
    ],
    { stdio: "inherit" },
  );

  if (validateRes.status !== 0) {
    console.error("❌ workspace-schema-validator 失敗；回退變更");
    rollback(name);
    process.exit(2);
  }

  updateWorkspacesReadme(name, answers);
  printPostGuide(name);
}

function rollback(name) {
  const wsFile = join(WORKSPACES_DIR, `${name}.md`);
  const wsDir = join(WORKSPACES_DIR, name);
  if (existsSync(wsFile)) rmSync(wsFile);
  if (existsSync(wsDir)) rmSync(wsDir, { recursive: true });
  // CURRENT_WORKSPACE 回退：若執行前不存在則 rm；若存在則寫回原值（需 Step 0 暫存）
}

// -------- Main --------
(async () => {
  try {
    const { schema, availableTechStacks } = await preCheck();
    const name =
      nameArg || (await rl.question("project_name [required, snake_case]: "));
    if (!name || !/^[a-z][a-z0-9_]*$/.test(name))
      fatal("project_name 格式錯誤（須 snake_case 英文）");

    const answers = await collectAnswers(schema, availableTechStacks);
    answers["PROJECT_CONTEXT.project_name"] = name;
    await previewAndCommit(name, answers);
  } finally {
    rl.close();
  }
})();
```

#### 1.5 測試要求（非新增 validator；僅 CLI 自測）

- **Dry-run 驗證**：`--dry-run` 模式執行完整流程不留檔；輸出預覽正確
- **回退驗證**：模擬 validator 失敗 → 確認 workspace 檔案與目錄完整回退
- **熔斷驗證**：輸入不存在的 tech-stacks → 觸發警告 + 三選一選單
- **既有 workspace 保護**：`CURRENT_WORKSPACE` 已存在且無 `--force` → 互動確認
- **schema 解析正確性**：手動改 SCHEMA 版本 → CLI 自動涵蓋新欄位問答

**不**新增自動化測試腳本（避免 CLI 互動性增加測試複雜度；依賴 CLI 自身 transaction-like 設計 + workspace-schema-validator 後置驗證）。

#### 1.6 禁止行為

- **絕對禁止** CLI 自動填入**無法由 User 確認**的值作為 required 欄位
- **絕對禁止** CLI 繞過 `workspace-schema-validator` 收尾驗證
- **絕對禁止** CLI 修改 `.agents/` 下 workspace 以外的檔案（rules / skills / tasks / tests 等不得動）
- **絕對禁止** CLI 留下半成品結構（失敗必須完整回退）
- **絕對禁止** CLI 呼叫網路 / 外部 API

1. **雙軌不取代**：CLI 與 AI 對話並存；User 擇優使用

5. **熔斷自動化**：對齊 `project-onboarding §Step 2` 但從人工升級為自動掃描
6. **驗證收尾**：結束必執行 `workspace-schema-validator`；失敗則回退
